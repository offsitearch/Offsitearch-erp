% ============================================================
% DATE-GROUP ROW MACROS
% ------------------------------------------------------------
% An employee can log multiple entries on the same date. These
% entries are grouped visually: the Date cell is merged (spans
% all entries for that date) and the whole group shares ONE
% background color. Consecutive date groups alternate between
% groupWhite and groupGray so each day is visually distinct.
%
% \dategroupfirst{color}{numEntries}{date}{hours}{project}{location}{description}
%   -> first (or only) row of a date group. Draws the merged date cell.
% \dategroupnext{color}{hours}{project}{location}{description}
%   -> additional entries for the SAME date (no date cell -- it's merged).
%
% Usage for a day with 3 entries (see ROWS section below for a
% full worked example):
%   \dategroupfirst{groupGray}{3}{01 Aug}{8.0}{Verde Residency}{Site -- 2nd Floor}{Coordinated ...}
%   \dategroupnext{groupGray}{7.5}{Verde Residency}{Studio}{Prepared ...}
%   \dategroupnext{groupGray}{3.0}{Internal}{Studio}{Updated ...}
%   \cmidrule(lr){1-5}
% Then flip to groupWhite for the next date, and so on.
% ============================================================
\newcommand{\dategroupfirst}[7]{%
  \rowcolor{#1}\multirow{#2}{*}{#3} & #4 & #5 & #6 & #7 \\
}
\newcommand{\dategroupnext}[5]{%
  \rowcolor{#1} & #2 & #3 & #4 & #5 \\
}
